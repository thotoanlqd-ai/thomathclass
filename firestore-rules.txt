// =========================================================
// LUẬT BẢO MẬT FIRESTORE — ThoMathClass
// =========================================================
// Cách dùng:
// 1. Vào Firebase Console > chọn project > Firestore Database > tab "Rules"
// 2. Xoá hết nội dung đang có, dán toàn bộ đoạn dưới đây vào
// 3. Thay CẢ HAI chỗ "ADMIN_UID_HERE" bằng UID tài khoản Giáo viên
//    (lấy ở Authentication > Users > bấm vào tài khoản của thầy)
// 4. Bấm "Publish" để áp dụng
// =========================================================

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // roster: danh sách công khai (chỉ tên + lớp), ai cũng xem được
    // nhưng CHỈ giáo viên mới được thêm/sửa/xoá
    match /roster/{studentId} {
      allow read: if true;
      allow write: if request.auth != null && request.auth.uid == "ADMIN_UID_HERE";
    }

    // students: dữ liệu riêng tư (điểm số, nhận xét)
    // chỉ chính học sinh đó (đã đăng nhập đúng tài khoản) hoặc giáo viên mới đọc được
    // chỉ giáo viên mới được ghi/sửa
    match /students/{studentId} {
      allow read: if request.auth != null &&
                     (request.auth.uid == studentId || request.auth.uid == "ADMIN_UID_HERE");
      allow write: if request.auth != null && request.auth.uid == "ADMIN_UID_HERE";
    }

    // products: nội dung Bài giảng / Tài liệu / Khóa học online / Bài tập Azota
    // ai cũng xem được (để hiện công khai trên web), chỉ giáo viên mới thêm/sửa/xoá
    match /products/{productId} {
      allow read: if true;
      allow write: if request.auth != null && request.auth.uid == "ADMIN_UID_HERE";
    }
  }
}
